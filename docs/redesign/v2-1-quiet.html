Done. A repeated request with the same key now replays the stored response instead of creating a second order.

export function idempotency(store: KeyStore) {
  return async (req, res, next) => {
    const key = req.get('Idempotency-Key');
    if (!key) return next();
    const hit = await store.get(key);
    if (hit) return res.status(hit.status).json(hit.body);
    res.locals.idempotencyKey = key;
    next();
  };
}

It lives in src/middleware/idempotency.ts and is mounted only on POST /orders.
